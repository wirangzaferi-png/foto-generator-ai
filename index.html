(function(){
  const splash=document.getElementById('wiranzaSplash');
  if(!splash) return;
  document.documentElement.classList.add('wz-loading');
  const closeSplash=()=>{
    splash.classList.add('hide');
    document.documentElement.classList.remove('wz-loading');
    setTimeout(()=>splash.remove(),650);
  };
  // Splash harus tampil minimal cukup lama untuk animasi logo selesai (~1.2 detik),
  // TAPI tidak boleh menambah jeda ekstra di atas waktu loading yang sudah lambat.
  // Dipicu oleh DOMContentLoaded (bukan window.load) supaya TIDAK ikut menunggu font/resource eksternal selesai diunduh.
  const START_TIME = Date.now();
  const MIN_SPLASH_MS = 1200;
  let closed = false;
  const closeOnce = ()=>{ if(closed) return; closed = true; closeSplash(); };
  const scheduleClose = ()=>{
    const elapsed = Date.now() - START_TIME;
    const remaining = Math.max(0, MIN_SPLASH_MS - elapsed);
    setTimeout(closeOnce, remaining);
  };
  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', scheduleClose, {once:true});
  } else {
    scheduleClose(); // dokumen sudah siap duluan (mis. script ini dimuat belakangan)
  }
  // Fallback keras: kalau ada resource yang macet total, splash tetap dijamin hilang, tidak mengunci aplikasi selamanya.
  setTimeout(closeOnce, 2000);
})();
